*=============================================
* FORTH.GFX - relocatable hi-res module
*
* Assembled at ORG 0 and loaded wherever the
* dictionary happens to start. Everything in
* the file is an offset from the load address,
* the ten byte header included, and MODLD adds
* that address to every word named in the fixup
* list at the end.
*
* Only addresses INSIDE this module are listed.
* Kernel routines go through the jump table at
* $0800, hardware addresses are fixed, and
* branches are relative - none of those move.
*=============================================
 xc                     ; 65C02 opcodes
*
*---- kernel jump table, frozen at $0800 ----
kpush equ $0803         ; A=hi Y=lo -> stack
kpop equ $0806          ; stack -> A=hi Y=lo
kpeek equ $0809
knext equ $080c         ; the inner interpreter
kdocol equ $080f
kemit equ $0812         ; print the char in A
*
*---- module communication block ----
* LOAD fills these in when it recognises a
* file called something.SHP. Both are zero
* when no shape file has been loaded.
shpadr equ $0825        ; where the shapes are
shpend equ $0827        ; first byte past them
*
*---- ROM and soft switches ----
pread equ $fb1e         ; X=paddle, returns Y
txtclr equ $c050        ; graphics
txtset equ $c051        ; text
mixclr equ $c052        ; full screen
mixset equ $c053        ; four lines of text
lowscr equ $c054        ; show page 1
hiscr equ $c055         ; show page 2
lores equ $c056         ; lo-res
hires equ $c057
btn0 equ $c061
btn1 equ $c062
btn2 equ $c063
*
*---- line tables, built by GFINIT ----
* Fixed addresses in RAM nothing else uses, so
* none of them needs a fixup, and 432 bytes of
* table cost the module nothing.
hrlo equ $7a00          ; 192 hi-res low bytes
hrhi equ $7ac0          ; 192 hi-res high bytes
lorlo equ $7b80         ; 24 lo-res low bytes
lorhi equ $7b98         ; 24 lo-res high bytes
*
*---- zero page ----
* Clear of the kernel ($06-$bd, $e0, $e5,
* $eb-$ee, $f0-$fc) and of the module loader
* ($be-$cb), which has finished by the time we
* run. Note the gaps at $e0 and $e5.
hptr equ $cc            ; screen address
hx equ $ce              ; current x, 16 bits
hy equ $d0              ; current y
hcolr equ $d1           ; colour byte
hmask equ $d2           ; pixel mask
hold equ $d3            ; the byte we found
hpg equ $d4             ; $20 or $40
gt2 equ $d5             ; divide working
gt3 equ $d7             ; byte offset
gt1 equ $d8             ; general pointer
*
* HLINE and DRAW share the rest. They are never
* active at the same time, so the storage is
* deliberately overlaid.
hx2 equ $da             ; HLINE: end x
hy2 equ $dc             ; HLINE: end y
hdx equ $dd             ; HLINE: delta x
hdy equ $df             ; HLINE: delta y
herr equ $e1            ; HLINE: error term
he2 equ $e3             ; HLINE: twice the error
hsx equ $e6             ; HLINE: x step
hsy equ $e7             ; HLINE: y step
shptr equ $da           ; DRAW: shape table
shght equ $dc           ; DRAW: rows left
shwid equ $dd           ; DRAW: bytes a row
shidx equ $de           ; DRAW: index in table
shcnt equ $df           ; SHAPE: shapes to skip
shsz equ $e1            ; SHAPE: size of one shape
*
* $e8 and $e9 used to hold the shape file's
* base address. LOAD keeps that in the
* communication block now, so they are free.
*
*---- fixup count ----
* How many DA lines the fixup list at the end
* of this file holds.  The module header below
* publishes it, and Merlin 8 resolves an EQU on
* its first pass - so the number has to be
* declared before the DA that reads it.  Add a
* fixup down there, add one to this.
nrel equ 143
*
 org 0

*=============================================
* MODRAW / MOXOR - one byte tells them apart
*
* EOR HCOLR ($45) gives
*   new = old EOR ((old EOR colour) AND mask)
* which replaces the pixel.
* LDA HCOLR ($A5) gives
*   new = old EOR (colour AND mask)
* which is XOR: draw twice and the background
* comes back untouched.
* Both are two-byte instructions, so only the
* opcode changes and nothing shifts. The same
* switch flips the shape plotter, where EOR
* ($51) becomes ORA ($11).
*=============================================
modraw lda #$45
x14 sta hmop
 lda #$11
x15 sta shsop
 rts
moxor lda #$a5
x16 sta hmop
 lda #$51
x17 sta shsop
 rts

 bne hc1

*=============================================
* SHDRAW - stamp the current shape at (HX,HY)
*
* Simoni's method (BYTE, August 1983): the
* shape is held the way the screen holds it, so
* plotting is a byte move rather than a walk
* through a vector list. A row is copied to a
* small buffer, shifted through the seven-bit
* boundaries to reach an x that is not a
* multiple of seven, then merged onto the line.
*
* Table: height, width in 7-bit groups (max 7),
* then height*width bytes. The index is one
* byte, so a table must stay under 256 bytes.
*=============================================
shdraw lda shptr
 ora shptr+1
 beq shret              ; no shape selected
x18 jsr hdiv7           ; GT3 = byte, GT2 = shift
 ldy #0
 lda (shptr),y
 sta shght
 iny
 lda (shptr),y
 sta shwid
 iny
 sty shidx
shrow lda hy
 cmp #192
 bcs shskip
* copy one row into the buffer, back to front
 ldx shwid
 ldy shidx
shcp lda (shptr),y
x19 sta shbuf,x
 iny
 dex
 bne shcp
 sty shidx
x20 stz shbuf           ; the byte bits spill into
* shift the whole row left GT2 times
 ldy gt2
 beq shpl
shs1 clc
 ldx shwid
 php                    ; keep the stack even
shs2 plp                ; carry from the last byte
x21 lda shbuf,x
 rol
 rol                    ; bit 6 falls into carry
 php                    ; save it for the next byte
 lsr                    ; back to seven bits
x22 sta shbuf,x
 dex
 cpx #$ff
 bne shs2
 plp
 dey
 bne shs1
* merge the row onto the screen
shpl equ *
x141 jsr haddr
 ldy gt3
 ldx shwid
shp1 equ *
x142 lda shbuf,x
shsop eor (hptr),y      ; patched: EOR or ORA
 sta (hptr),y
 iny
 dex
 cpx #$ff
 bne shp1
 bra shnxt
shskip clc              ; off screen: skip the row
 lda shidx
 adc shwid
 sta shidx
shnxt inc hy
 dec shght
 bne shrow
shret rts
shbuf ds 8

*=============================================
* HGR ( n -- ) a hi-res screen, cleared
*
* 1 HGR and 2 HGR pick the page: $2000 or
* $4000. Page 1 comes up mixed, so the prompt
* stays visible along the bottom. PAGE 2 COMES
* UP FULL SCREEN AND CANNOT BE MIXED.
*
* There is one page select for the whole
* display, not one for graphics and another
* for text. So the four text lines at the foot
* of a mixed screen are read from text page 2
* whenever the graphics are read from hi-res
* page 2 - and text page 2 is $0800, which is
* where this system lives. A mixed page 2 puts
* the kernel on the screen as characters.
*
* This is exactly why Applesoft's HGR was
* mixed and its HGR2 full screen. A BASIC
* program started at $0801 and would have been
* displayed as garbage for the same reason. It
* was not an arbitrary pairing; it was the
* hardware.
*=============================================
gffirst da 0            ; MODLD patches this
 dfb 3
 asc "HGR"
x23 da w_hgr
w_hgr jsr kpop
 tya
 cmp #2
 beq w_hgr2
 lda #$20
 sta hpg
x24 jsr hclr
 lda #1
x25 sta gdpg
 sta txtclr
 sta mixset             ; page 1 may be mixed
 sta hires
 sta lowscr
 jmp knext
w_hgr2 lda #$40
 sta hpg
x26 jsr hclr
 lda #2
x27 sta gdpg
 sta txtclr
 sta mixclr             ; page 2 NEVER mixed
 sta hires              ; hi-res BEFORE page 2
 sta hiscr
 jmp knext

*=============================================
* SHAPE ( n -- ) choose the nth shape
*
* No index table is needed. Every shape begins
* with its height and width, so shape n is
* found by stepping over the n before it, each
* 2 + height*width bytes long. A shape file is
* therefore nothing but tables laid end to end
* - concatenate what the editor gives you and
* it works.
*
* The file itself arrives through LOAD, which
* recognises it by its name and records where
* it put it in the communication block. So
* there is nothing to tell SHAPE: it looks the
* address up. Asking for a shape the file does
* not contain gives shape 0 rather than a walk
* off the end of it, and asking with no file
* loaded says so.
*=============================================
x55 da h_hscr
 dfb 5
 asc "SHAPE"
x56 da w_shap
w_shap jsr kpop
 sty shcnt
 lda shpadr
 sta shptr
 lda shpadr+1
 sta shptr+1
 ora shptr              ; both zero: no file
 beq shsno
shsl lda shcnt
 beq shsx
 dec shcnt
 ldy #0
 lda (shptr),y          ; height
 sta shght
 iny
 lda (shptr),y          ; width
 sta shwid
 lda #2                 ; size = 2 + height*width
 sta shsz
 stz shsz+1
 ldx shwid
 beq shsa2
shsa1 clc
 lda shsz
 adc shght
 sta shsz
 bcc shsa0
 inc shsz+1

 bne shsa1
shsa2 clc
 lda shptr
 adc shsz
 sta shptr
 lda shptr+1
 adc shsz+1
 sta shptr+1
 lda shptr+1            ; still inside the file?
 cmp shpend+1
 bcc shsl
 bne shsrst
 lda shptr
 cmp shpend
 bcc shsl
shsrst lda shpadr       ; past the end: shape 0
 sta shptr
 lda shpadr+1
 sta shptr+1
shsx jmp knext
shsno lda #$3f ;"?"
 jsr kemit
 stz shptr
 stz shptr+1

*=============================================
* THE LO-RES SCREEN
*
* 40 across by 48 down, in the same memory the
* text screen uses. Each byte holds two blocks
* stacked one above the other: the low nibble
* is the even row, the high nibble the odd one.
* So a row address serves two rows, and the
* table only needs 24 entries.
*
* Row by row through the table, never a bulk
* fill: the 80-column firmware keeps state in
* the holes between the rows, and writing over
* them jams the cursor.
*=============================================
*
*---- lorow - point GT1 at row A's line ----
* out: A = 0 for the low nibble, 1 for the high
lorow pha
 lsr                    ; two rows to a line
 tax
 lda lorlo,x
 sta gt1
 lda lorhi,x
 sta gt1+1
 pla
 and #1
 rts
*
*---- loput - colour the block at column Y ----
* Y is preserved, so a row can be walked with
* nothing but INY.
loput lda (gt1),y
x64 ldx ghalf
 bne lopu1
 and #$f0               ; keep the odd row
x65 ora gcol
 bra lopu2
lopu1 and #$0f          ; keep the even row
x66 ora gcolh
lopu2 sta (gt1),y
 rts

*---- GR ( -- ) the lo-res screen, cleared ----
x67 da h_btn
 dfb 2
 asc "GR"
x68 da w_gr
* Forty columns first. The lo-res screen IS
* the text screen, so with the 80-column
* firmware running, half of every row would be
* coming out of auxiliary memory and the
* picture would be shredded. Ctrl-Q is the
* same code TEXT40 uses, and it does no harm
* if 80 columns were never turned on - GRCLR
* wipes the screen straight afterwards.
w_gr lda #$11 ;ctrl-Q
 jsr kemit
x69 jsr grclr
 lda #1                 ; there is no lo-res page 2
x70 sta gdpg
 sta lowscr             ; page 1 FIRST - coming from
 sta txtclr             ; page 2, anything else would
 sta mixset             ; show $0800 as lo-res first
 sta lores
 jmp knext
h_gr equ x67

*
*=============================================
* SHOW ( n -- ) display a page without
* clearing it
*
* HPAGE says where to DRAW, SHOW says what to
* LOOK AT, and the two together are what makes
* animation without flicker: build the next
* picture on the page nobody is watching, then
* SHOW it. HGR would clear the page first,
* which is the one thing that must not happen.
*
* Showing page 2 turns the mixing off, for the
* same reason HGR does.
*=============================================
x136 da h_mixd
 dfb 4
 asc "SHOW"
x137 da w_show
w_show jsr kpop
 tya
 cmp #2
 beq w_shw2
 lda #1
x138 sta gdpg
 sta hires
 sta lowscr
 jmp knext
w_shw2 lda #2
x139 sta gdpg
 sta mixclr             ; before, never after
 sta hires
 sta hiscr
 jmp knext
h_show equ x136
